from openpyxl import Workbook
from openpyxl.styles import Font, PatternFill

dias = [("01/09", 35.0, 23.4, 1.7), ("02/09", 36.5, 23.7, 0.7),
        ("03/09", 38.0, 25.0, 0.2), ("04/09", 34.1, 25.2, 0.7),
        ("05/09", 32.2, 23.6, 6.9), ("06/09", 30.2, 22.3, 0.7)]
libro = Workbook()
hoja = libro.active
hoja.title = "Clima"
hoja.append(["Día", "Máx °C", "Mín °C", "Lluvia mm"])
for fila in dias:
    hoja.append(fila)
AZUL = PatternFill(fill_type="solid", start_color="00408F")
for celda in hoja[1]:
    celda.font = Font(bold=True, color="FFFFFF")
    celda.fill = AZUL
for fila in hoja.iter_rows(min_row=2, min_col=2):
    for celda in fila:
        celda.number_format = "0.0"
hoja.column_dimensions["D"].width = 12
hoja.freeze_panes = "A2"
hoja.auto_filter.ref = hoja.dimensions
libro.save("clima_formato.xlsx")

from datetime import date
from openpyxl import Workbook
crudos = [("2026-09-01", "35.0"), ("2026-09-31", "34.0"),
          ("2026-09-04", "N/D"), ("2026-09-05", "32.2")]
libro = Workbook()
datos = libro.active
datos.title = "Datos"
errores = libro.create_sheet("Errores")
datos.append(["Fecha", "Máxima °C"])
errores.append(["Fecha", "Máxima", "Motivo"])
for f, t in crudos:
    try:
        fila = [date.fromisoformat(f), float(t)]
    except ValueError as e:
        errores.append([f, t, str(e)])
        continue
    datos.append(fila)
datos.column_dimensions["A"].width = 12
errores.column_dimensions["A"].width = 12
errores.column_dimensions["C"].width = 36
libro.save("limpio.xlsx")
print("Válidos:", datos.max_row - 1)
print("Con error:", errores.max_row - 1)

import statistics as st
from openpyxl import Workbook
from openpyxl.styles import Font

tmax = [35.0, 36.5, 38.0, 34.1, 32.2, 30.2, 34.0]

def tabla(hoja, encabezados, filas):
    hoja.append(encabezados)
    for fila in filas:
        hoja.append(fila)
    for celda in hoja[1]:
        celda.font = Font(bold=True)

libro = Workbook()
datos = libro.active
datos.title = "Datos"
tabla(datos, ["Día", "Máxima °C"], enumerate(tmax, start=1))
resumen = [("Media", round(st.mean(tmax), 2)),
           ("Mediana", st.median(tmax)),
           ("Desviación", round(st.stdev(tmax), 2))]
tabla(libro.create_sheet("Resumen"), ["Medida", "°C"], resumen)
libro.save("reporte_semana.xlsx")
