import statistics as st
from fractions import Fraction

tmax = [35.0, 36.5, 38.0, 34.1, 32.2, 30.2, 34.0, 33.3, 34.9, 35.1,
        37.2, 35.0, 35.2, 35.8, 35.9, 35.0, 34.7, 33.5, 34.5, 35.4,
        35.4, 32.8, 31.8, 31.5, 33.3, 33.9, 34.2, 33.7, 33.2, 30.9]
lluvia = [1.7, 0.7, 0.2, 0.7, 6.9, 0.7, 0.0, 0.3, 0.6, 0.8,
          0.2, 0.8, 0.2, 0.2, 0.0, 0.0, 1.9, 2.7, 0.0, 0.0,
          0.0, 2.9, 3.2, 0.5, 0.9, 0.7, 2.7, 2.6, 0.3, 8.8]

def resumen(datos):
    return {"media": st.mean(datos), "mediana": st.median(datos),
            "desv": st.stdev(datos)}

r = resumen(tmax)
print(f"Media {r['media']:.1f} °C, mediana {r['mediana']:.2f} °C")
print(f"Desviación estándar: {r['desv']:.2f} °C")
dias = len([x for x in lluvia if x >= 1.0])
p = Fraction(dias, len(lluvia))
print(f"Llovió {dias} de {len(lluvia)} días: {p} = {float(p):.0%}")
